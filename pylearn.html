What is Python?

Python is a high-level, interpreted, dynamically-typed language designed for readability. It is the #1 language for data engineering, data science, and AI due to its vast ecosystem (NumPy, Pandas, PyTorch, Airflow), clean syntax, and the ability to move quickly from prototype to production.

Functions

        
Reusable blocks of code. The backbone of maintainable data pipelines.

      

    


    

      
Definition

      
A function is a named, reusable block of code that performs a specific task. In data engineering, functions encapsulate transformations, validations, and API calls — making pipelines modular and testable.

    


    

      
functions.py

      
# Basic function with type hints (always use in production)
def clean_string(text: str, lower: bool = True) -> str:
    """Strip whitespace and optionally lowercase a string."""
    result = text.strip()
    return result.lower() if lower else result

print(clean_string("  Hello World  "))   # hello world
print(clean_string("  Hello  ", lower=False))  # Hello

def total(*numbers: float) -> float:
    return sum(numbers)

# **kwargs — variable keyword arguments
def build_query(**filters) -> str:
    clauses = [f"{k} = '{v}'" for k, v in filters.items()]
    return "WHERE " + " AND ".join(clauses)

print(build_query(status="active", country="US"))
# WHERE status = 'active' AND country = 'US'

# Returning multiple values (as tuple)
def stats(data: list) -> tuple:
    return min(data), max(data), sum(data)/len(data)

lo, hi, avg = stats([1, 5, 3, 9])

Decorators


    

      
Definition: Decorator

      
A decorator is a function that wraps another function to extend its behavior without modifying it. Used heavily in data engineering for logging, timing, retrying API calls, and caching.

    


    

      
decorators.py

      
import time
import functools

# Timer decorator — measure pipeline step performance
def timer(func):
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        start = time.perf_counter()
        result = func(*args, **kwargs)
        elapsed = time.perf_counter() - start
        print(f"{func.__name__} took {elapsed:.4f}s")
        return result
    return wrapper

@timer
def process_batch(rows):
    return [r.upper() for r in rows]

process_batch(["a", "b", "c"])
# process_batch took 0.0001s

# Retry decorator — essential for API calls & DB connections
def retry(max_attempts=3, delay=1):
    def decorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            for attempt in range(max_attempts):
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    if attempt == max_attempts - 1:
                        raise
                    time.sleep(delay)
            return wrapper
    return decorator

@retry(max_attempts=3, delay=2)
def fetch_api_data(url: str) -> dict:
    pass  # Will retry 3 times with 2s delay on failure

# Built-in: functools.lru_cache — memoize expensive functions
from functools import lru_cache

@lru_cache(maxsize=128)
def expensive_lookup(key: str) -> str:
    # DB call, API call, etc. — cached after first call
    return key.upper()

Classes, inheritance, dataclasses, and dunder methods — structure your data pipelines like a pro.

from dataclasses import dataclass, field
from typing import List, Optional
from abc import ABC, abstractmethod

# ── Basic Class ──
class DataPipeline:
    # Class variable (shared across all instances)
    version: str = "1.0.0"

    def __init__(self, name: str, schedule: str = "@daily"):
        # Instance variables
        self.name = name
        self.schedule = schedule
        self._runs: list = []  # private by convention

    def run(self) -> str:
        result = f"[{self.name}] started"
        self._runs.append(result)
        return result

    @property
    def run_count(self) -> int:   # property = computed attribute
        return len(self._runs)

    @classmethod
    def from_config(cls, config: dict) -> "DataPipeline":
        return cls(config["name"], config.get("schedule", "@daily"))

    @staticmethod
    def validate_schedule(schedule: str) -> bool:
        return schedule.startswith("@") or schedule.startswith("0 ")

    def __repr__(self):
        return f"DataPipeline(name={self.name!r}, schedule={self.schedule!r})"

    def __str__(self):
        return f"Pipeline '{self.name}' ({self.schedule})"

# ── Inheritance ──
class ETLPipeline(DataPipeline):
    def __init__(self, name, source, target):
        super().__init__(name)      # call parent __init__
        self.source = source
        self.target = target

    def run(self):
        print(f"Extract from {self.source}")
        print(f"Load into {self.target}")
        return super().run()     # call parent method

# ── Abstract Base Class ── (interface pattern)
class Connector(ABC):
    @abstractmethod
    def connect(self) -> None: ...
    @abstractmethod
    def fetch(self, query: str) -> list: ...

class PostgresConnector(Connector):
    def connect(self): print("Connected to Postgres")
    def fetch(self, query): return []   # impl here

# ── Dataclass (Python 3.7+) — best for data records ──
@dataclass
class Event:
    event_id: str
    user_id: int
    event_type: str
    properties: dict = field(default_factory=dict)
    processed: bool = False

    def is_conversion(self) -> bool:
        return self.event_type in {"purchase", "signup", "upgrade"}

e = Event("evt_001", 42, "purchase", {"amount": 99.99})
print(e.is_conversion())  # True
print(e)  # Event(event_id='evt_001', user_id=42, ...)

try/except, custom exceptions, logging — essential for robust, observable pipelines.

      

    


    

      
error_handling.py

      
import logging

# ── Logging setup (do this in every module) ──
logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s | %(levelname)s | %(name)s | %(message)s",
    datefmt="%Y-%m-%d %H:%M:%S"
)
log = logging.getLogger(__name__)

# ── try / except / else / finally ──
try:
    result = 10 / 0
except ZeroDivisionError as e:
    log.error(f"Division error: {e}")
    result = None
except (TypeError, ValueError) as e:
    log.warning(f"Type/value error: {e}")
    result = 0
else:
    log.info("Calculation succeeded")   # only if no exception
finally:
    log.debug("Cleanup runs always")    # always runs (close connections!)

# ── Custom exceptions ── (define a hierarchy)
class PipelineError(Exception):
    """Base exception for all pipeline errors."""

class ValidationError(PipelineError):
    def __init__(self, field: str, value, message: str):
        self.field = field
        self.value = value
        super().__init__(f"Field '{field}' failed: {message} (got {value!r})")

class ConnectionError(PipelineError): ...
class TransformError(PipelineError): ...

# Raise and catch custom exceptions
def validate_age(age):
    if not isinstance(age, int) or age < 0:
        raise ValidationError("age", age, "must be a non-negative int")
    return age

try:
    validate_age(-5)
except ValidationError as e:
    log.error(str(e))   # Field 'age' failed: must be non-negative (got -5)

# ── Context manager for resource cleanup ──
class DatabaseConn:
    def __enter__(self):
        log.info("DB connected"); return self
    def __exit__(self, exc_type, exc_val, exc_tb):
        log.info("DB closed")
        return False  # don't suppress exceptions

with DatabaseConn() as db:
    print("Running query...")

# Generator function — yield one item at a time (memory efficient!)
def read_chunks(filepath: str, chunk_size: int = 1000):
    """Stream a large file in chunks — never loads all at once."""
    with open(filepath) as f:
        chunk = []
        for line in f:
            chunk.append(line)
            if len(chunk) >= chunk_size:
                yield chunk    # pause and return chunk
                chunk = []      # reset
        if chunk:
            yield chunk        # final partial chunk

# Use it — processes 100M rows with minimal RAM

GIL — The Python Gotcha

      
Python's Global Interpreter Lock (GIL) prevents true thread parallelism for CPU-bound code. Rule of thumb: use asyncio for I/O-bound tasks (API calls, DB queries), use multiprocessing for CPU-bound tasks (heavy computation), use threading for simple I/O parallelism.

# ── Production ETL Pattern ──
import logging
from dataclasses import dataclass
from typing import Iterator
import pandas as pd

log = logging.getLogger(__name__)

@dataclass
class PipelineConfig:
    source_path: str
    target_table: str
    chunk_size: int = 10_000
    drop_nulls: bool = True

def extract(config: PipelineConfig) -> Iterator[pd.DataFrame]:
    """Extract data in chunks — handles files of any size."""
    for chunk in pd.read_csv(config.source_path, chunksize=config.chunk_size):
        log.info(f"Extracted chunk: {len(chunk)} rows")
        yield chunk

def transform(df: pd.DataFrame, config: PipelineConfig) -> pd.DataFrame:
    """Clean, validate, and enrich data."""
    if config.drop_nulls:
        df = df.dropna()
    df.columns = [c.strip().lower().replace(" ", "_") for c in df.columns]
    df["processed_at"] = pd.Timestamp.now(tz="UTC")
    return df

def load(df: pd.DataFrame, engine, table: str, batch: int) -> None:
    """Load data — append mode for incremental pipelines."""
    df.to_sql(table, con=engine, if_exists="append", index=False, chunksize=500)
    log.info(f"Loaded batch {batch}: {len(df)} rows → {table}")

def run_etl(config: PipelineConfig, engine) -> dict:
    """Orchestrate the full ETL."""
    stats = {"total_rows": 0, "batches": 0, "errors": 0}
    for i, raw_chunk in enumerate(extract(config)):
        try:
            clean = transform(raw_chunk, config)
            load(clean, engine, config.target_table, i)
            stats["total_rows"] += len(clean)
            stats["batches"] += 1
        except Exception as e:
            log.error(f"Batch {i} failed: {e}")
            stats["errors"] += 1
    return stats

# ── Airflow DAG skeleton ──
from airflow import DAG
from airflow.operators.python import PythonOperator
from datetime import datetime, timedelta

with DAG(
    dag_id="user_events_etl",
    schedule="0 2 * * *",          # daily at 2am
    start_date=datetime(2024, 1, 1),
    catchup=False,
    default_args={"retries": 3, "retry_delay": timedelta(minutes=5)}
) as dag:

    extract_task = PythonOperator(task_id="extract", python_callable=extract)
    transform_task = PythonOperator(task_id="transform", python_callable=transform)
    load_task = PythonOperator(task_id="load", python_callable=load)

    extract_task >> transform_task >> load_task

Datetime & Timezones

        
The #1 source of silent bugs in data pipelines. Get this right every time.

      

    

    

      
Golden Rule

      
Always store timestamps in UTC. Convert to local time only at presentation. A pipeline that silently processes data for the wrong day due to timezone miscalculation can corrupt months of reporting.

PySpark & Big Data

        
Process terabytes across a cluster — the industry standard for large-scale data engineering.

      

    

    

      
Spark vs Pandas

      
Use Pandas for data up to ~1GB in RAM. Use PySpark when data exceeds memory, you need distributed processing, or you're working with a data lakehouse (Delta Lake, Iceberg). Spark is the backbone of most enterprise data platforms.

    

    

      
pyspark_etl.py

      
from pyspark.sql import SparkSession
from pyspark.sql import functions as F
from pyspark.sql.window import Window

spark = (SparkSession.builder
    .appName("EventsPipeline")
    .config("spark.sql.adaptive.enabled", "true")
    .getOrCreate())

# Read (lazy — nothing runs yet)
df = spark.read.parquet("s3://bucket/events/year=2024/*")

# Transformations — chain with method syntax
cleaned = (df
    .filter(F.col("event_type").isin(["purchase", "signup"]))
    .withColumn("event_date", F.to_date("event_ts"))
    .dropna(subset=["user_id"])
    .drop_duplicates(["event_id"])
)

# Aggregation
agg = cleaned.groupBy("event_date", "country").agg(
    F.sum("revenue").alias("total_revenue"),
    F.countDistinct("user_id").alias("unique_users"),
)

# Window functions
w = Window.partitionBy("user_id").orderBy("event_ts")
df_w = cleaned.withColumn("row_num", F.row_number().over(w))

# Spark SQL alternative
cleaned.createOrReplaceTempView("events")
result = spark.sql("SELECT event_date, SUM(revenue) FROM events GROUP BY 1")

# Write — triggers actual computation
agg.write.mode("overwrite").partitionBy("event_date").parquet("s3://bucket/processed/")
agg.write.format("delta").mode("append").save("/mnt/delta/events")

spark.stop()

Data Quality Dimensions

      
Completeness (no unexpected nulls), Accuracy (values in valid ranges), Consistency (same format across sources), Timeliness (data arrives when expected), Uniqueness (no unexpected duplicates). Validate these in your pipeline to prevent garbage-in, garbage-out.

from pydantic import BaseModel, Field, field_validator, model_validator, ValidationError
from typing import Optional
from datetime import datetime

class Event(BaseModel):
    event_id:   str           = Field(..., min_length=1)
    user_id:    int           = Field(..., gt=0)
    event_type: str
    revenue:    Optional[float] = Field(None, ge=0)
    event_ts:   datetime

    @field_validator("event_type")
    @classmethod
    def validate_type(cls, v):
        allowed = {"click", "view", "purchase", "signup"}
        if v not in allowed: raise ValueError(f"Must be one of {allowed}")
        return v

    @model_validator(mode="after")
    def revenue_required_for_purchase(self):
        if self.event_type == "purchase" and self.revenue is None:
            raise ValueError("revenue required for purchase")
        return self
